# CDP Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a CDP mode that suspends ABP and starts Chrome's remote debugging server, allowing external CDP tools to take full browser control, then cleanly returns control to ABP on exit.

**Architecture:** Extends the existing `InputMode` enum with `kCdp`. Entering CDP mode reuses human mode's suspend logic (abort actions, save execution state, resume tabs, allow system inputs), then detaches ABP's CDP clients and starts Chrome's `DevToolsAgentHost::StartRemoteDebuggingServer()` with auto-port selection. Exiting stops the server, re-attaches CDP clients to all tabs (including new ones), and restores execution state. Toolbar icon shows a purple plug icon in CDP mode.

**Tech Stack:** C++ (Chromium DevTools, net::TCPServerSocket, content::DevToolsAgentHost), ABP REST API, MCP JSON-RPC, Chromium PageActionIconView, Skia vector icons.

**Design Spec:** `docs/plans/2026-03-26-cdp-mode-design.md`

int AbpController::FindAvailableCdpPort(int start_port, int max_attempts) {
  for (int i = 0; i < max_attempts; ++i) {
    int port = start_port + i;
    auto socket = std::make_unique<net::TCPServerSocket>(nullptr, net::NetLogSource());
    net::IPEndPoint endpoint(net::IPAddress::IPv4Localhost(), port);
    int rv = socket->Listen(endpoint, 1 /* backlog */, std::nullopt);
    if (rv == net::OK) {
      // Socket binds successfully — port is available.
      // Socket is destroyed here, releasing the port.
      return port;
    }
  }
  return -1;  // No available port found.
}
```

void AbpController::ReattachAllCdpClients() {
  DCHECK_CURRENTLY_ON(content::BrowserThread::UI);

  // Re-start event observer — Stop() detaches all event clients,
  // Start() re-discovers all tabs via DevToolsAgentHost::GetOrCreateAll().
  if (event_observer_) {
    event_observer_->Stop();
    event_observer_->Start();
  }

  // Collect all current WebContents from TabStripModel.
  std::set<std::string> live_tab_ids;
  const BrowserList* browser_list = BrowserList::GetInstance();
  for (auto it = browser_list->begin(); it != browser_list->end(); ++it) {
    Browser* browser = *it;
    TabStripModel* tab_strip = browser->tab_strip_model();
    for (int i = 0; i < tab_strip->count(); ++i) {
      content::WebContents* wc = tab_strip->GetWebContentsAt(i);
      if (!wc) continue;

      auto host = content::DevToolsAgentHost::GetOrCreateFor(wc);
      const std::string& tab_id = host->GetId();
      live_tab_ids.insert(tab_id);

      // Create CDP client (attaches to DevToolsAgentHost).
      // GetOrCreateCdpClient also enables Runtime.enable, Page.enable.
      GetOrCreateCdpClient(wc);

      // For tabs created by the external CDP client (no prior ABP state),
      // enable default execution control if it's globally enabled.
      if (!tab_states_.count(tab_id) || !tab_states_[tab_id].execution.IsEnabled()) {
        if (IsExecutionControlEnabled()) {
          // GetOrCreateCdpClient already created the TabState entry.
          // Enable execution control for this new tab.
          EnableExecutionControl(tab_id, base::DoNothing());
        }
      }
    }
  }

  // Clean up stale tab_states_ entries for tabs closed during CDP mode.
  std::vector<std::string> stale_ids;
  for (auto& [tab_id, state] : tab_states_) {
    if (!live_tab_ids.contains(tab_id)) {
      stale_ids.push_back(tab_id);
    }
  }
  for (const auto& tab_id : stale_ids) {
    tab_states_.erase(tab_id);
  }
}
```

Note: `GetOrCreateCdpClient` handles CDP client creation and base domain enablement. `event_observer_->Stop()` + `Start()` is the correct re-attach pattern — `AttachToHost` is private, but `Start()` calls it for all existing hosts. New tabs created by the external CDP client get default execution control via `EnableExecutionControl` if globally enabled.

```cpp
void AbpController::EnterCdpMode(const base::Value::Dict& params,
                                  ResponseCallback callback) {
  DCHECK_CURRENTLY_ON(content::BrowserThread::UI);

  if (input_mode_ == InputMode::kCdp) {
    SendError(409, "already in cdp mode", std::move(callback));
    return;
  }

  // 1. Determine port.
  int requested_port = params.FindInt("port").value_or(0);
  int port;
  if (requested_port > 0) {
    // Explicit port — test it.
    port = FindAvailableCdpPort(requested_port, 1);
  } else {
    // Auto-select starting at 24578.
    port = FindAvailableCdpPort(24578, 100);
  }
  if (port < 0) {
    SendError(503, "no available port", std::move(callback));
    return;
  }

  // 2. If in agent mode, do the full suspend (same as SwitchToHumanMode).
  if (input_mode_ == InputMode::kAgent) {
    // Abort in-flight actions.
    for (auto& [tab_id, state] : tab_states_) {
      AbortActiveAction(tab_id);
    }

    // Save execution state and resume paused tabs.
    saved_execution_states_.clear();
    for (auto& [tab_id, state] : tab_states_) {
      if (state.execution.IsEnabled()) {
        SavedExecutionState saved;
        saved.was_enabled = true;
        saved.was_paused = state.execution.IsPaused();
        saved.virtual_time_base_ms = state.execution.virtual_time_base_ticks_ms;
        saved_execution_states_[tab_id] = saved;

        if (state.execution.IsPaused()) {
          ResumeExecution(tab_id, base::DoNothing());
        }
      }
    }

    // Allow system inputs.
    SetAllowSystemInputsForAllTabs(true);
  }
  // If in human mode, execution is already suspended and system inputs allowed.

  // 3. Detach ABP's CDP clients.
  DetachAllCdpClients();

  // 4. Start remote debugging server.
  content::DevToolsAgentHost::StartRemoteDebuggingServer(
      std::make_unique<AbpCdpSocketFactory>(port),
      base::FilePath(),  // No port output file.
      base::FilePath(),  // No custom frontend.
      content::DevToolsAgentHost::RemoteDebuggingServerMode::kDefault);

  // 5. Read back actual address and construct ws_url.
  cdp_port_ = port;
  std::string address =
      content::DevToolsAgentHost::GetRemoteDebuggingServerAddress();
  // address is "host:port" format. Construct WebSocket URL with browser target path.
  // The /devtools/browser path is the standard CDP browser target endpoint.
  cdp_ws_url_ = "ws://" + address + "/devtools/browser";

  // 6. Start timeout timer if requested.
  std::optional<int> timeout_ms = params.FindInt("timeout_ms");
  if (timeout_ms.has_value() && *timeout_ms > 0) {
    cdp_timeout_deadline_ =
        base::TimeTicks::Now() + base::Milliseconds(*timeout_ms);
    cdp_timeout_timer_.Start(
        FROM_HERE, base::Milliseconds(*timeout_ms),
        base::BindOnce(&AbpController::OnCdpModeTimeout,
                        base::Unretained(this)));
  } else {
    cdp_timeout_deadline_ = base::TimeTicks();
  }

  // 7. Update state and notify observers.
  input_mode_ = InputMode::kCdp;
  for (auto& observer : input_mode_observers_) {
    observer.OnInputModeChanged(InputMode::kCdp);
  }

  // 8. Build response.
  base::Value::Dict response;
  response.Set("status", "ok");
  response.Set("port", cdp_port_);
  response.Set("ws_url", cdp_ws_url_);
  if (timeout_ms.has_value() && *timeout_ms > 0) {
    response.Set("timeout_ms", *timeout_ms);
  }
  SendJson(200, base::Value(std::move(response)), std::move(callback));
}
```

```cpp
void AbpController::ExitCdpMode(ResponseCallback callback) {
  DCHECK_CURRENTLY_ON(content::BrowserThread::UI);

  if (input_mode_ != InputMode::kCdp) {
    SendError(409, "not in cdp mode", std::move(callback));
    return;
  }

  // 1. Stop remote debugging server.
  content::DevToolsAgentHost::StopRemoteDebuggingServer();
  cdp_port_ = 0;
  cdp_ws_url_.clear();

  // 2. Cancel timeout timer.
  cdp_timeout_timer_.Stop();
  cdp_timeout_deadline_ = base::TimeTicks();

  // 3. Temporarily set mode to kAgent so GetOrCreateCdpClient guard allows re-attach.
  input_mode_ = InputMode::kAgent;

  // 4. Re-attach CDP clients to all tabs (discovers new tabs from TabStripModel).
  ReattachAllCdpClients();

  // 5. Block system inputs and restore execution state (same as SwitchToAgentMode).
  SetAllowSystemInputsForAllTabs(false);

  for (auto& [tab_id, saved] : saved_execution_states_) {
    if (saved.was_enabled && saved.was_paused) {
      auto it = tab_states_.find(tab_id);
      if (it != tab_states_.end() && it->second.execution.IsEnabled()) {
        PauseExecution(tab_id, base::DoNothing());
      }
    }
  }
  saved_execution_states_.clear();

  // 6. Notify observers (mode already set to kAgent above).
  for (auto& observer : input_mode_observers_) {
    observer.OnInputModeChanged(InputMode::kAgent);
  }

  base::Value::Dict response;
  response.Set("status", "ok");
  SendJson(200, base::Value(std::move(response)), std::move(callback));
}
```

void AbpMcpHandler::CallCdpMode(const base::Value::Dict& args,
                                 base::Value request_id,
                                 ResponseWithHeadersCallback callback) {
  const std::string* action = args.FindString("action");
  if (!action) {
    SendToolError("Missing required parameter: action",
                  std::move(request_id), std::move(callback));
    return;
  }

  if (*action == "enter") {
    // Pass args through — EnterCdpMode reads "port" and "timeout_ms" from it.
    std::string body;
    base::JSONWriter::Write(base::Value(args.Clone()), &body);
    controller_->HandleRequest(
        "POST", "/api/v1/browser/cdp-mode/enter", body,
        base::BindOnce(&AbpMcpHandler::OnControllerResponse,
                       weak_factory_.GetWeakPtr(), std::move(request_id),
                       std::move(callback)));
  } else if (*action == "exit") {
    controller_->HandleRequest(
        "POST", "/api/v1/browser/cdp-mode/exit", "",
        base::BindOnce(&AbpMcpHandler::OnControllerResponse,
                       weak_factory_.GetWeakPtr(), std::move(request_id),
                       std::move(callback)));
  } else {
    SendToolError("Invalid action: must be 'enter' or 'exit'",
                  std::move(request_id), std::move(callback));
  }
}

Update `OnExecuting()` to handle CDP mode click (exits CDP mode):

```cpp
void AbpInputModeIconView::OnExecuting(ExecuteSource source) {
  abp::AbpController* controller = abp::AbpController::GetInstance();
  if (!controller) {
    return;
  }

  abp::AbpController::InputMode actual_mode = controller->GetInputMode();

  if (actual_mode == abp::AbpController::InputMode::kCdp) {
    // Clicking in CDP mode exits back to agent mode.
    controller->ExitCdpMode(
        base::BindOnce([](int, const std::string&, std::string) {}));
    return;
  }

  // Toggle between agent and human mode.
  base::Value::Dict params;
  if (actual_mode == abp::AbpController::InputMode::kAgent) {
    params.Set("input_mode", "human");
  } else {
    params.Set("input_mode", "agent");
  }
  controller->SetInputMode(params, base::DoNothing());
}
```
